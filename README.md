## 2. Реализация программы
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

#include <math.h>

#define PI 3.14159265

int main() {

	double x, y, z, t;
	printf("Введите значение х: ");
	scanf("%lf", &x);

	printf("Введите значение y: ");
	scanf("%lf", &y);

	printf("Введите значение z: ");
	scanf("%lf", &z);

	double ver1 = 2 * cos(x - PI / 6);
	double niz1 = 0.5 + pow(sin(y) , 2);
	double drob1 = ver1 / niz1;

	double drob2 = (z * z) / (3 - (z * z) / 5);
	double S = 1 + drob2;

	t = drob1 * S;

	printf("Результат t = %f\n", t);

	return 0;
}
```
